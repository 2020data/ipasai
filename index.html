/**
 * 每日五題 40 天成長小徑 - Google Apps Script 安全後端 (Code.gs)
 * 密碼完全存放於 GAS「專案設定 → 指令碼屬性 → ADMIN_PASSWORD」，絕不寫入前端程式碼。
 *
 * 自動初始化三個工作表：
 * 1. Visits (username, count, lastDate, completedDays)
 * 2. Errors (questionId, questionText, totalErrors, wrongA, wrongB, wrongC, wrongD)
 * 3. Bank   (key, value)
 */

function getAdminPassword() {
  return PropertiesService.getScriptProperties().getProperty('ADMIN_PASSWORD') || '';
}

function verifyPassword(inputPwd) {
  const realPwd = getAdminPassword();
  if (!realPwd) {
    return {
      ok: false,
      authFailed: true,
      error: '尚未在 GAS 指令碼屬性中設定 ADMIN_PASSWORD，請至「專案設定 → 指令碼屬性」新增。'
    };
  }
  if (typeof inputPwd !== 'string' || inputPwd !== realPwd) {
    return { ok: false, authFailed: true, error: '管理者密碼錯誤，請重新輸入。' };
  }
  return { ok: true };
}

function getOrInitSheet_(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (headers && headers.length) {
      sheet.appendRow(headers);
      sheet.setFrozenRows(1);
    }
  } else if (headers && headers.length) {
    // 若既有工作表缺少新欄位（如 completedDays），自動補齊表頭
    const lastCol = Math.max(sheet.getLastColumn(), 1);
    const existingHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    if (existingHeaders.length < headers.length) {
      for (let c = existingHeaders.length; c < headers.length; c++) {
        sheet.getRange(1, c + 1).setValue(headers[c]);
      }
    }
  }
  return sheet;
}

function initAllSheets_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const visitsSheet = getOrInitSheet_(ss, 'Visits', ['username', 'count', 'lastDate', 'completedDays']);
  const errorsSheet = getOrInitSheet_(ss, 'Errors', ['questionId', 'questionText', 'totalErrors', 'wrongA', 'wrongB', 'wrongC', 'wrongD']);
  const bankSheet   = getOrInitSheet_(ss, 'Bank',   ['key', 'value']);
  return { visitsSheet, errorsSheet, bankSheet };
}

function parseDaysJson_(raw) {
  if (!raw) return [];
  try {
    const arr = JSON.parse(String(raw));
    if (Array.isArray(arr)) {
      return arr.map(Number).filter(n => n >= 1 && n <= 40).sort((a, b) => a - b);
    }
  } catch (e) {}
  return [];
}

function jsonOut_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * GET: 讀取題庫、來訪與 40 天進度、錯題統計與 hasAdminPassword 狀態
 */
function doGet(e) {
  try {
    const { visitsSheet, errorsSheet, bankSheet } = initAllSheets_();

    // 1. 讀取 Visits（含唯一帳號 40 天完成進度）
    const vRows = visitsSheet.getDataRange().getValues();
    const visits = [];
    for (let i = 1; i < vRows.length; i++) {
      const name = String(vRows[i][0] || '').trim();
      if (!name) continue;
      visits.push({
        username: name,
        count: Number(vRows[i][1] || 0),
        lastDate: String(vRows[i][2] || ''),
        completedDays: parseDaysJson_(vRows[i][3])
      });
    }

    // 2. 讀取 Errors
    const eRows = errorsSheet.getDataRange().getValues();
    const errors = [];
    for (let i = 1; i < eRows.length; i++) {
      const qId = String(eRows[i][0] || '').trim();
      if (!qId) continue;
      errors.push({
        questionId: qId,
        questionText: String(eRows[i][1] || ''),
        totalErrors: Number(eRows[i][2] || 0),
        wrongA: Number(eRows[i][3] || 0),
        wrongB: Number(eRows[i][4] || 0),
        wrongC: Number(eRows[i][5] || 0),
        wrongD: Number(eRows[i][6] || 0)
      });
    }

    // 3. 讀取 Bank
    const bRows = bankSheet.getDataRange().getValues();
    let bank = null;
    for (let i = 1; i < bRows.length; i++) {
      if (String(bRows[i][0]) === 'bank_json' && bRows[i][1]) {
        try {
          bank = JSON.parse(bRows[i][1]);
        } catch (err) {
          bank = null;
        }
        break;
      }
    }

    return jsonOut_({
      ok: true,
      bank: bank,
      visits: visits,
      errors: errors,
      hasAdminPassword: Boolean(getAdminPassword())
    });
  } catch (err) {
    return jsonOut_({ ok: false, error: String(err) });
  }
}

/**
 * POST: 處理 loginUser / recordDayComplete / recordVisit / recordError / verifyAdmin / saveBank / resetStats
 */
function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    const payload = JSON.parse((e.postData && e.postData.contents) || '{}');
    const action = payload.action;

    // 1. 管理者登入驗證
    if (action === 'verifyAdmin') {
      return jsonOut_(verifyPassword(payload.password));
    }

    const { visitsSheet, errorsSheet, bankSheet } = initAllSheets_();

    // 2. 跨裝置唯一帳號登入／自動建帳
    if (action === 'loginUser') {
      const username = String(payload.username || '').trim();
      const dateStr  = String(payload.date || Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy-MM-dd'));
      if (!username) return jsonOut_({ ok: false, error: 'Missing username' });

      const vValues = visitsSheet.getDataRange().getValues();
      for (let i = 1; i < vValues.length; i++) {
        if (String(vValues[i][0]).trim() === username) {
          return jsonOut_({
            ok: true,
            isNew: false,
            user: {
              username: username,
              count: Number(vValues[i][1] || 0),
              lastDate: String(vValues[i][2] || ''),
              completedDays: parseDaysJson_(vValues[i][3])
            }
          });
        }
      }
      // 尚未存在則自動建立唯一帳號
      visitsSheet.appendRow([username, 0, dateStr, '[]']);
      return jsonOut_({
        ok: true,
        isNew: true,
        user: { username: username, count: 0, lastDate: dateStr, completedDays: [] }
      });
    }

    // 3. 紀錄來訪與 40 天完成進度（支援重複考試累計次數，並保留所有已完成天數）
    if (action === 'recordDayComplete' || action === 'recordVisit') {
      const username = String(payload.username || '').trim();
      const dateStr  = String(payload.date || Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy-MM-dd'));
      const dayNum   = Number(payload.day || 0);
      if (!username) return jsonOut_({ ok: false, error: 'Missing username' });

      const vValues = visitsSheet.getDataRange().getValues();
      let foundRow = -1;
      for (let i = 1; i < vValues.length; i++) {
        if (String(vValues[i][0]).trim() === username) {
          foundRow = i + 1;
          break;
        }
      }

      if (foundRow > 0) {
        const prevCount = Number(vValues[foundRow - 1][1] || 0);
        const days = parseDaysJson_(vValues[foundRow - 1][3]);
        if (dayNum >= 1 && dayNum <= 40 && days.indexOf(dayNum) === -1) {
          days.push(dayNum);
          days.sort((a, b) => a - b);
        }
        visitsSheet.getRange(foundRow, 2, 1, 3).setValues([[prevCount + 1, dateStr, JSON.stringify(days)]]);
        return jsonOut_({
          ok: true,
          action: action,
          user: { username: username, count: prevCount + 1, lastDate: dateStr, completedDays: days }
        });
      } else {
        const initDays = (dayNum >= 1 && dayNum <= 40) ? [dayNum] : [];
        visitsSheet.appendRow([username, 1, dateStr, JSON.stringify(initDays)]);
        return jsonOut_({
          ok: true,
          action: action,
          user: { username: username, count: 1, lastDate: dateStr, completedDays: initDays }
        });
      }
    }

    // 4. 紀錄單題匿名錯誤選項（不綁定任何用戶個資）
    if (action === 'recordError') {
      const qId = String(payload.questionId || '').trim();
      const qText = String(payload.questionText || '');
      const chosenKey = String(payload.chosenKey || '').toUpperCase();
      if (!qId) return jsonOut_({ ok: false, error: 'Missing questionId' });

      const eValues = errorsSheet.getDataRange().getValues();
      let foundRow = -1;
      for (let i = 1; i < eValues.length; i++) {
        if (String(eValues[i][0]).trim() === qId) {
          foundRow = i + 1;
          break;
        }
      }

      const incA = chosenKey === 'A' ? 1 : 0;
      const incB = chosenKey === 'B' ? 1 : 0;
      const incC = chosenKey === 'C' ? 1 : 0;
      const incD = chosenKey === 'D' ? 1 : 0;

      if (foundRow > 0) {
        const row = eValues[foundRow - 1];
        errorsSheet.getRange(foundRow, 2, 1, 6).setValues([[
          qText || String(row[1] || ''),
          Number(row[2] || 0) + 1,
          Number(row[3] || 0) + incA,
          Number(row[4] || 0) + incB,
          Number(row[5] || 0) + incC,
          Number(row[6] || 0) + incD
        ]]);
      } else {
        errorsSheet.appendRow([qId, qText, 1, incA, incB, incC, incD]);
      }
      return jsonOut_({ ok: true, action: 'recordError' });
    }

    // 5. 儲存題庫名稱與題目（需後端密碼複驗）
    if (action === 'saveBank') {
      const auth = verifyPassword(payload.password);
      if (!auth.ok) return jsonOut_(auth);

      const bankObj = {
        title: String(payload.title || '每日五題・40 天成長小徑'),
        subtitle: String(payload.subtitle || ''),
        questions: Array.isArray(payload.questions) ? payload.questions : []
      };
      const bValues = bankSheet.getDataRange().getValues();
      let foundRow = -1;
      for (let i = 1; i < bValues.length; i++) {
        if (String(bValues[i][0]) === 'bank_json') {
          foundRow = i + 1;
          break;
        }
      }
      const serialized = JSON.stringify(bankObj);
      if (foundRow > 0) {
        bankSheet.getRange(foundRow, 2).setValue(serialized);
      } else {
        bankSheet.appendRow(['bank_json', serialized]);
      }
      return jsonOut_({ ok: true, action: 'saveBank' });
    }

    // 6. 重置統計數據（需後端密碼複驗，保留題庫不變）
    if (action === 'resetStats') {
      const auth = verifyPassword(payload.password);
      if (!auth.ok) return jsonOut_(auth);

      if (visitsSheet.getLastRow() > 1) {
        visitsSheet.deleteRows(2, visitsSheet.getLastRow() - 1);
      }
      if (errorsSheet.getLastRow() > 1) {
        errorsSheet.deleteRows(2, errorsSheet.getLastRow() - 1);
      }
      return jsonOut_({ ok: true, action: 'resetStats' });
    }

    return jsonOut_({ ok: false, error: 'Unknown action' });
  } catch (err) {
    return jsonOut_({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}
